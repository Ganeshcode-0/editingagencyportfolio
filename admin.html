Auto-Sync to your Google Sheets

Want new editor submissions to automatically appear in your Google Sheet? (Click to view 2-minute setup)

▼

Step 1: In your Google Drive, open or create a Google Sheet named "GanaCuts Visuals — Agency Editors".

Step 2: Go to Extensions > Apps Script, replace the code with the script below, and click Deploy > New deployment > Web app (Access: Anyone):

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(),
    data.name || '',
    data.email || '',
    data.phone || '',
    data.specialization || '',
    data.tools || '',
    data.rate || '',
    data.portfolio || '',
    data.status || 'pending',
    data.notes || ''
  ]);
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
